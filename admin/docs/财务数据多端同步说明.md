# 财务数据多端同步功能说明

**版本**: v1.0  
**更新日期**: 2025-01-XX  
**适用范围**: Web管理后台 + 小程序店长管理后台

---

## 📋 功能概述

实现了财务数据的云端存储和多端同步功能，支持：
- ✅ Web管理后台 ↔ 云数据库双向同步
- ✅ 小程序店长管理后台 ↔ 云数据库双向同步
- ✅ 自动同步（数据变更时）
- ✅ 手动同步（用户触发）
- ✅ 冲突解决（时间戳优先策略）
- ✅ 离线支持（本地缓存）

## 🗄️ 数据库集合

### 1. finance_records（财务账本记录）

**集合结构**：
```javascript
{
  _id: String,              // 文档ID
  date: String,             // 日期（YYYY-MM-DD格式）
  revenue: Number,          // 营收金额
  orderCount: Number,       // 订单数量
  // 其他自定义字段...
  _createTime: Number,       // 创建时间戳
  _updateTime: Number,       // 更新时间戳
  syncedAt: Number          // 同步时间戳
}
```

**索引建议**：
- `date`: 唯一索引（确保同一天只有一条记录）

### 2. finance_settings（财务设置）

**集合结构**：
```javascript
{
  _id: String,              // 文档ID
  settings: {               // 设置对象
    target: Number,          // 目标营收
    invest: Number,          // 投资金额
    costRate: Number,        // 成本率
    rent: Number,            // 租金
    staffCount: Number,      // 员工数量
    staffSalary: Number,     // 员工薪资
    // 其他设置...
  },
  _createTime: Number,       // 创建时间戳
  _updateTime: Number,       // 更新时间戳
  syncedAt: Number          // 同步时间戳
}
```

## 📱 小程序端集成指南

### 1. 创建云函数（可选，推荐）

在小程序项目中创建云函数 `syncFinanceData`：

```javascript
// cloudfunctions/syncFinanceData/index.js
const cloud = require('wx-server-sdk')
cloud.init({ env: cloud.DYNAMIC_CURRENT_ENV })
const db = cloud.database()

exports.main = async (event, context) => {
  const wxContext = cloud.getWXContext()
  const { action, records, settings } = event
  
  try {
    if (action === 'upload_records') {
      // 批量更新或新增记录
      const batch = []
      for (const record of records) {
        const existing = await db.collection('finance_records')
          .where({ date: record.date })
          .get()
        
        if (existing.data && existing.data.length > 0) {
          batch.push(
            db.collection('finance_records')
              .doc(existing.data[0]._id)
              .update({
                ...record,
                _updateTime: Date.now(),
                syncedAt: Date.now()
              })
          )
        } else {
          batch.push(
            db.collection('finance_records')
              .add({
                ...record,
                _createTime: Date.now(),
                _updateTime: Date.now(),
                syncedAt: Date.now()
              })
          )
        }
      }
      
      await Promise.all(batch)
      return { success: true, message: `成功同步 ${records.length} 条记录` }
    }
    
    if (action === 'download_records') {
      const res = await db.collection('finance_records')
        .orderBy('date', 'desc')
        .limit(1000)
        .get()
      return { success: true, data: res.data }
    }
    
    if (action === 'upload_settings') {
      const existing = await db.collection('finance_settings')
        .limit(1)
        .get()
      
      const settingsData = {
        settings: settings,
        _updateTime: Date.now(),
        syncedAt: Date.now()
      }
      
      if (existing.data && existing.data.length > 0) {
        await db.collection('finance_settings')
          .doc(existing.data[0]._id)
          .update(settingsData)
      } else {
        settingsData._createTime = Date.now()
        await db.collection('finance_settings')
          .add(settingsData)
      }
      
      return { success: true, message: '设置同步成功' }
    }
    
    if (action === 'download_settings') {
      const res = await db.collection('finance_settings')
        .orderBy('_updateTime', 'desc')
        .limit(1)
        .get()
      
      return { 
        success: true, 
        data: res.data && res.data.length > 0 ? res.data[0] : null 
      }
    }
    
    return { success: false, message: '未知操作' }
  } catch (error) {
    console.error('同步失败:', error)
    return { success: false, message: error.message }
  }
}
```

### 2. 在小程序中创建同步工具

创建 `utils/financeSync.js`：

```javascript
// utils/financeSync.js
const db = wx.cloud.database()

const FINANCE_RECORDS_KEY = 'shanhai_v12'
const FINANCE_SETTINGS_KEY = 'sc_finance_v12'

/**
 * 从云端获取财务记录
 */
export async function fetchFinanceRecordsFromCloud() {
  try {
    const res = await wx.cloud.callFunction({
      name: 'syncFinanceData',
      data: { action: 'download_records' }
    })
    
    if (res.result.success) {
      return res.result.data || []
    }
    throw new Error(res.result.message || '获取失败')
  } catch (error) {
    console.error('从云端获取财务记录失败:', error)
    throw error
  }
}

/**
 * 上传财务记录到云端
 */
export async function saveFinanceRecordsToCloud(records) {
  try {
    const res = await wx.cloud.callFunction({
      name: 'syncFinanceData',
      data: {
        action: 'upload_records',
        records: records
      }
    })
    
    if (res.result.success) {
      return res.result
    }
    throw new Error(res.result.message || '上传失败')
  } catch (error) {
    console.error('上传财务记录失败:', error)
    throw error
  }
}

/**
 * 同步财务记录（双向同步）
 */
export async function syncFinanceRecords() {
  try {
    // 1. 读取本地数据
    const localRaw = wx.getStorageSync(FINANCE_RECORDS_KEY)
    const localRecords = localRaw ? JSON.parse(localRaw) : []
    
    // 2. 从云端获取数据
    const cloudRecords = await fetchFinanceRecordsFromCloud()
    
    // 3. 合并数据（优先使用最新的）
    const merged = mergeRecords(localRecords, cloudRecords)
    
    // 4. 保存到本地
    wx.setStorageSync(FINANCE_RECORDS_KEY, JSON.stringify(merged))
    
    // 5. 上传到云端
    await saveFinanceRecordsToCloud(merged)
    
    return { success: true, mergedCount: merged.length }
  } catch (error) {
    console.error('同步失败:', error)
    return { success: false, error: error.message }
  }
}

/**
 * 合并记录（解决冲突）
 */
function mergeRecords(localRecords, cloudRecords) {
  const merged = []
  const recordMap = new Map()
  
  // 先添加云端记录
  cloudRecords.forEach(record => {
    recordMap.set(record.date, {
      ...record,
      updateTime: record._updateTime || record.syncedAt || 0
    })
  })
  
  // 合并本地记录（如果本地更新则覆盖）
  localRecords.forEach(record => {
    const existing = recordMap.get(record.date)
    const localUpdateTime = record._updateTime || 0
    
    if (!existing || localUpdateTime > existing.updateTime) {
      recordMap.set(record.date, {
        ...record,
        updateTime: localUpdateTime
      })
    }
  })
  
  // 转换为数组并排序
  recordMap.forEach((record, date) => {
    merged.push({
      date: record.date,
      revenue: record.revenue || 0,
      orderCount: record.orderCount || 0,
      ...record
    })
  })
  
  merged.sort((a, b) => new Date(a.date) - new Date(b.date))
  return merged
}
```

### 3. 在店长管理后台页面集成

在 `pages/manager/index.js` 中：

```javascript
// pages/manager/index.js
import { syncFinanceRecords } from '../../utils/financeSync'

Page({
  data: {
    syncing: false
  },
  
  onLoad() {
    // 页面加载时自动同步
    this.autoSync()
  },
  
  // 自动同步
  async autoSync() {
    try {
      const result = await syncFinanceRecords()
      if (result.success) {
        console.log('财务数据同步成功')
      }
    } catch (error) {
      console.warn('自动同步失败:', error)
    }
  },
  
  // 手动同步
  async handleSync() {
    this.setData({ syncing: true })
    wx.showLoading({ title: '同步中...', mask: true })
    
    try {
      const result = await syncFinanceRecords()
      if (result.success) {
        wx.showToast({ title: '同步成功', icon: 'success' })
        // 刷新页面数据
        this.loadFinanceData()
      } else {
        wx.showToast({ title: '同步失败', icon: 'none' })
      }
    } catch (error) {
      wx.showToast({ title: '同步失败：' + error.message, icon: 'none' })
    } finally {
      this.setData({ syncing: false })
      wx.hideLoading()
    }
  },
  
  // 保存财务数据时自动同步
  saveFinanceData(data) {
    // 保存到本地
    wx.setStorageSync('shanhai_v12', JSON.stringify(data))
    
    // 异步同步到云端
    syncFinanceRecords().catch(err => {
      console.warn('自动同步失败:', err)
    })
  }
})
```

## 🔄 同步策略

### 冲突解决策略

1. **时间戳优先**
   - 比较记录的 `_updateTime` 或 `syncedAt`
   - 使用更新时间最新的记录

2. **本地优先**（可选）
   - 如果本地有更新，优先使用本地数据
   - 适用于离线场景

3. **云端优先**（可选）
   - 强制使用云端数据
   - 适用于数据恢复场景

### 同步时机

1. **自动同步**
   - 页面加载时
   - 数据保存时
   - 定时同步（可选，如每5分钟）

2. **手动同步**
   - 用户点击同步按钮
   - 下拉刷新时

## 📊 数据流程

### Web管理后台

```
用户操作 → 保存到localStorage → 自动上传到云端
         ↓
    页面加载 → 从云端同步 → 更新localStorage → 显示数据
```

### 小程序端

```
用户操作 → 保存到Storage → 调用云函数上传
         ↓
    页面加载 → 调用云函数下载 → 更新Storage → 显示数据
```

## ⚠️ 注意事项

1. **数据一致性**
   - 确保日期格式统一（YYYY-MM-DD）
   - 确保金额和数量字段类型一致

2. **性能优化**
   - 批量操作使用 Promise.all
   - 大量数据时考虑分页同步

3. **错误处理**
   - 网络失败时使用本地数据
   - 同步失败时记录错误日志

4. **权限设置**
   - 数据库集合权限：所有用户可读，仅创建者可写
   - 或使用云函数统一处理权限
