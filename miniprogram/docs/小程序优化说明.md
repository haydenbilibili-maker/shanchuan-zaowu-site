## 🔧 七、使用指南

### 7.1 错误监控使用
```javascript
const errorMonitor = require('../../utils/errorMonitor');

// 捕获错误
errorMonitor.captureError({
  type: 'customError',
  level: 'error',
  message: '错误信息',
  stack: '错误堆栈'
});

// 捕获 API 错误
errorMonitor.captureApiError({
  url: '/api/data',
  method: 'GET',
  status: 500,
  message: '服务器错误'
});
```

### 7.2 性能监控使用
```javascript
const performanceMonitor = require('../../utils/performance');

// 监控页面加载
performanceMonitor.monitorPageLoad('index');

// 监控 API 调用
const result = await performanceMonitor.monitorApi('getData', async () => {
  return await wx.cloud.callFunction({ name: 'getData' });
});
```

### 7.3 图片加载使用
```javascript
const imageLoader = require('../../utils/imageLoader');

// 优化图片 URL
const optimizedUrl = imageLoader.optimizeUrl(url, {
  width: 750,
  quality: 80,
  format: 'webp'
});

// 预加载图片
await imageLoader.preload(url);

// 批量预加载
await imageLoader.preloadBatch([url1, url2, url3]);
```

### 7.4 智能缓存使用
```javascript
const { smartCache, CACHE_EXPIRY } = require('../../utils/cache');

// 智能缓存（优先返回缓存，后台更新）
const data = await smartCache('BANNERS', async () => {
  return await fetchBanners();
}, CACHE_EXPIRY.BANNERS);
```
